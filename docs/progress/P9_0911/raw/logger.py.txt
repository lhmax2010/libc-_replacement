import sys, subprocess, datetime, pathlib, json
root=pathlib.Path('/home/toolchain/development/libcxx_runtime_validation/docs/progress/P9_0911/raw')
label, command=sys.argv[1:3]
prefix=root/label
prefix.with_suffix('.command.txt').write_text(command+'\n')
start=datetime.datetime.now(datetime.timezone.utc).isoformat()
with prefix.with_suffix('.stdout').open('wb') as out, prefix.with_suffix('.stderr').open('wb') as err:
    p=subprocess.run(command,shell=True,executable='/bin/bash',stdout=out,stderr=err)
prefix.with_suffix('.exitcode').write_text(str(p.returncode)+'\n')
prefix.with_suffix('.time.json').write_text(json.dumps({'start':start,'end':datetime.datetime.now(datetime.timezone.utc).isoformat()},indent=2)+'\n')
print('COMMAND:',command)
print('EXIT:',p.returncode)
print(prefix.with_suffix('.stdout').read_text(errors='replace'))
print(prefix.with_suffix('.stderr').read_text(errors='replace'),file=sys.stderr)
sys.exit(p.returncode)
