ORIG = {r['snapshot']:r['original_path'] for r in csv.DictReader((ROOT/'SOURCE_MANIFEST.tsv').open(),delimiter='\t')}
def link(key):
    f,p,r=S[key]
    return f'[{p}:{r[0][0]}](sources/{f}/{p}.txt#L{r[0][0]})'
def resolve(text):
    return re.sub(r'@([a-z]+)@',lambda m:link(m[1]),text)
EXCERPTS=[]
def blocks(keys, lang):
    result=[]
    for key in keys:
        family,path,ranges=S[key]
        snap=f'sources/{family}/{path}.txt'
        lines=(ROOT/snap).read_text().splitlines(keepends=True)
        original=ORIG[snap]
        for a,b in ranges:
            assert 0<a<=b<=len(lines),(key,a,b,len(lines))
            body=''.join(lines[a-1:b])
            result.append(f'[{original}:{a}–{b}]({snap}#L{a})\n\n```cpp\n{body.rstrip(chr(10))}\n```\n')
            EXCERPTS.append(dict(id=key,family=family,path=path,start=a,end=b,sha256=hashlib.sha256(body.encode()).hexdigest()))
    return '\n'.join(result)
