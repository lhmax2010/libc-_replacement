import urllib.request,json
from pathlib import Path
urls=["https://gcc.gnu.org/onlinedocs/libstdc%2B%2B/manual/abi.html","https://gcc.gnu.org/onlinedocs/libstdc%2B%2B/manual/using_dual_abi.html","https://gcc.gnu.org/pipermail/libstdc%2B%2B/2016-July/044437.html"]
rows=[]
for url in urls:
 try:
  with urllib.request.urlopen(url,timeout=20) as f: status=f.status
  rows.append({"url":url,"http_status":status})
 except Exception as e: rows.append({"url":url,"http_status":"NOT_AVAILABLE","error":str(e)})
print(json.dumps(rows,indent=2))
Path("docs/progress/P9_0911/GNU_URL_RETRY.json").write_text(json.dumps(rows,indent=2)+"\n")
