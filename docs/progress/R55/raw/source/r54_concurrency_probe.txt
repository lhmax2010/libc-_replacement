    41	int cancel_and_join_all(pthread_t* threads, int count, bool repeat) {
    42	  int failures = 0;

    47	    failures += first != 0;
    48	    failures += repeat && second != 0;
    49	  }
    50	  for (int i = 0; i < count; ++i) {
    51	    void* result = nullptr;
    52	    const int rc = pthread_join(threads[i], &result);

    54	                 i, rc, result == PTHREAD_CANCELED, result);

    56	  }
