"""Read one build's current progress without changing its state."""
import datetime,json,sys
from pathlib import Path
p=Path('progress/BPF_W1_0921/cells')/sys.argv[1]
print(datetime.datetime.now().astimezone().isoformat())
lines=(p/'build.log').read_text(errors='replace').splitlines()
items=[x for x in lines if x.startswith(('[','Wrote:','error:','COPY_AND_','NORMAL_TRANSACTION_'))]
print('\n'.join(items[-4:]))
rc=p/'exitcode'; print('EXITCODE '+rc.read_text().strip() if rc.exists() else 'RUNNING')
events=p/'events.jsonl'
if events.exists():
    rows=[json.loads(x) for x in events.read_text().splitlines() if x.strip()]
    probes=[x for x in rows if x.get('event')=='io_probe']
    pauses=[x for x in rows if x.get('event') in ('paused','stop','pause_retry_failed')]
    print('I/O probes',len(probes),'pause/stop events',len(pauses))
    if probes: print('last probe seconds',probes[-1]['seconds'],'exitcode',probes[-1]['exitcode'])
