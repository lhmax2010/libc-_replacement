set -euo pipefail
workspace=/home/toolchain/development/libc++_replacement
target=$workspace/docs/progress/R64
if test -e "$target"; then
  echo "TARGET_ALREADY_EXISTS=$target" >&2
  exit 1
fi
mkdir -p "$target/code/evidence" "$target/code/input"
cp "$workspace/progress/R64/REPORT.md" "$target/REPORT.md"
cp "$workspace/progress/R64/COMMAND_EVIDENCE_LIMITATIONS.md" "$target/code/"
cp -a "$workspace/progress/R64/code/." "$target/code/"
cp -a "$workspace/progress/R64/commands" "$target/code/evidence/"
cp -a "$workspace/progress/R64/raw" "$target/code/evidence/"
cp -a "$workspace/progress/R64/tables" "$target/code/evidence/"
cp "$workspace/docs/progress/W2/code/tests/r61_m2_matrix.cpp" "$target/code/input/r61_m2_matrix.original.cpp"
printf 'source\tuploaded_path\tsha256\n' > "$target/code/INDEX.tsv"
find "$target" -type f ! -path "$target/code/INDEX.tsv" -print0 | sort -z | while IFS= read -r -d '' file; do
  relative=${file#"$workspace/"}
  hash=$(sha256sum "$file" | awk '{print $1}')
  printf '%s\t%s\t%s\n' "$relative" "$relative" "$hash"
done >> "$target/code/INDEX.tsv"
find "$target" -type f -size +5M -printf 'OVERSIZE\t%p\t%s\n'
find "$target" -type f -printf '%p\t%s\n' | sort
