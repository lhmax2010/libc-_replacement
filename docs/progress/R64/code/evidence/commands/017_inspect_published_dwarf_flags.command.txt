set -o pipefail
app_debug=tmp/R64/w3-published-debug/app-installers/usr/lib/debug/usr/lib/libapp-installers.so.1.37.2.debug
tpk_debug=tmp/R64/w3-published-debug/tpk-installer/usr/lib/debug/usr/lib/libtpk-installer.so.debug
sha256sum "$app_debug" "$tpk_debug"
for f in "$app_debug" "$tpk_debug"; do
  printf '=== SECTIONS %s ===\n' "$f"
  readelf -SW "$f" | rg 'GCC.command|debug_info|debug_str|debug_line'
  printf '=== PRODUCERS %s ===\n' "$f"
  readelf --debug-dump=info "$f" | rg 'DW_AT_producer' | sort -u
  printf '=== FLAG STRINGS %s ===\n' "$f"
  strings "$f" | rg -- '-flto|-fno-lto|CMAKE_BUILD_TYPE| -O[0-3s] |LTO' | sort -u | head -200
  printf 'FLAG_STRING_PIPE_EXIT=%s,%s,%s\n' "${PIPESTATUS[0]}" "${PIPESTATUS[1]}" "${PIPESTATUS[2]}"
done
