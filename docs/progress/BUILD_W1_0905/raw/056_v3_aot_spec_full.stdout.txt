# Dependency provenance policy: everything installable from the Tizen
# Base/Unified repos comes via BuildRequires (gcc, python3, zip, unzip,
# patchelf, ...). Only three things cannot (no bazel package exists; Tizen
# python3 is 3.14 while TF 2.18 requires <=3.12): the bazel binary, the
# hermetic CPython 3.12, and the pinned pip wheels. Each vendored binary is
# runtime-verified inside this Tizen buildroot below and the build fails
# fast if any of them cannot execute here.
./bazel --version || { echo "FATAL: vendored bazel cannot run in this buildroot"; exit 1; }

export HERMETIC_PYTHON_VERSION=%{py_ver}
# Keep bazel's output base outside the source tree: %%prep wipes the tree on
# every run, so an in-tree output base means a full ~4h rebuild after even a
# one-line spec change. The buildroot itself survives between gbs runs.
export BAZEL_ROOT=$HOME/.cache/bazel-tf-aot
mkdir -p "$BAZEL_ROOT"

# Force gcc for bazel's C++ toolchain autodetection (see BuildRequires note).
export CC=/usr/bin/gcc
export CXX=/usr/bin/g++
export BAZEL_COMPILER=gcc

# Offline pip: rules_python's whl_library shells out to pip to resolve the
# requirements lock. Disable pip isolation so it honors these env vars and
# point it at the bundled wheel mirror instead of PyPI.
export RULES_PYTHON_PIP_ISOLATED=0
export PIP_NO_INDEX=1
export PIP_FIND_LINKS=$PWD/_pip_mirror

# Tizen coreutils ships no realpath(1); TF's toolchain-probing repository
# rules call it via `bash -c "realpath ..."`. Provide a shim.
mkdir -p $PWD/shims
printf '#!/bin/sh\nexec readlink -f "$@"\n' > $PWD/shims/realpath
# ... and no timeout(1) either; minimal shell replacement (rc!=0 on expiry).
cat > $PWD/shims/timeout << 'TMOEOF'
#!/bin/sh
secs=$1; shift
"$@" & pid=$!
( t=0; while [ $t -lt $secs ]; do sleep 10; t=$((t+10)); kill -0 $pid 2>/dev/null || exit 0; done; kill -9 $pid 2>/dev/null ) & wd=$!
wait $pid; rc=$?
kill $wd 2>/dev/null; wait $wd 2>/dev/null
exit $rc
TMOEOF
chmod +x $PWD/shims/realpath $PWD/shims/timeout
export PATH=$PWD/shims:$PATH

# Resource caps: TF/XLA translation units peak at 2-3GB each with gcc;
# unbounded parallelism has frozen the build host before. Cap both the
# action count and the memory budget bazel schedules against. Raise with
# --define "aot_build_jobs N" on a dedicated machine.
%if "%{aot_arch}" == "aarch64"
# The bazel JVM under qemu-user sporadically deadlocks during analysis with
# no output (observed locally and on the build farm, where an 8h inactivity
# watchdog then kills the job). Warm the analysis phase under a timeout and
# restart the server on a hang: turns a dead build into a <=90min retry.
for attempt in 1 2 3; do
    if timeout 5400 ./bazel --output_user_root=$BAZEL_ROOT \
        --host_jvm_args=-XX:ActiveProcessorCount=2 \
        build --nobuild \
        -c opt --curses=no --color=no \
        --copt=-Wno-error=incompatible-pointer-types \
        --jobs=%{aot_build_jobs} \
        --local_cpu_resources=%{aot_build_jobs} \
        --local_ram_resources=8192 \
        --spawn_strategy=local \
        --repository_cache=$PWD/_repo_cache \
        --experimental_repository_disable_download \
        //tensorflow/tools/pip_package:wheel \
        //tensorflow/compiler/aot:tfcompile \
        //tensorflow/python/tools:saved_model_cli; then
        echo "analysis warm-up OK (attempt $attempt)"
        break
    fi
    echo "analysis attempt $attempt hung or failed; restarting bazel server"
    timeout 120 ./bazel --output_user_root=$BAZEL_ROOT shutdown || true
    for f in $BAZEL_ROOT/*/server/server.pid.txt; do
        [ -f "$f" ] && kill -9 "$(cat "$f")" 2>/dev/null || true
    done
    [ $attempt = 3 ] && { echo "FATAL: bazel analysis kept hanging"; exit 1; }
done
%endif
./bazel --output_user_root=$BAZEL_ROOT \
%if "%{aot_arch}" == "aarch64"
    --host_jvm_args=-XX:ActiveProcessorCount=2 \
%endif
    build \
    -c opt \
    --curses=no --color=no --verbose_failures \
%if "%{aot_arch}" == "aarch64"
    --copt=-Wno-error=incompatible-pointer-types \
%endif
    --jobs=%{aot_build_jobs} \
    --local_cpu_resources=%{aot_build_jobs} \
    --local_ram_resources=8192 \
    --spawn_strategy=local \
    --repository_cache=$PWD/_repo_cache \
    --experimental_repository_disable_download \
    //tensorflow/tools/pip_package:wheel \
    //tensorflow/compiler/aot:tfcompile \
    //tensorflow/python/tools:saved_model_cli

# Shut the server down so nothing holds files open during %%install.
./bazel --output_user_root=$BAZEL_ROOT shutdown

%install
AOT_ROOT=%{buildroot}%{aot_prefix}
SITE_DIR=$AOT_ROOT/lib/python%{py_ver}/site-packages
mkdir -p $AOT_ROOT/bin $SITE_DIR
# %%install runs in a fresh shell; mirror %%build's output base.
BAZEL_ROOT=$HOME/.cache/bazel-tf-aot

# 1. Bundled hermetic CPython.
PY_EXT=$(echo $BAZEL_ROOT/*/external/python_%{py_platform})
test -d "$PY_EXT" || { echo "FATAL: hermetic python dir not found"; exit 1; }
cp -a "$PY_EXT" $AOT_ROOT/python
rm -f $AOT_ROOT/python/BUILD.bazel $AOT_ROOT/python/WORKSPACE
# Runtime-verify the bundled interpreter inside this Tizen buildroot.
$AOT_ROOT/python/bin/python%{py_ver} -c "import sys; print(sys.version)" || \
    { echo "FATAL: bundled CPython cannot run in this buildroot"; exit 1; }

# 2. TensorFlow wheel -> site-packages.
WHEEL=$(find bazel-bin/tensorflow/tools/pip_package -name "tensorflow-%{version}*.whl" | head -1)
test -n "$WHEEL" || { echo "FATAL: built tensorflow wheel not found"; exit 1; }
unzip -q "$WHEEL" -d $SITE_DIR

# 3. Python runtime deps of saved_model_cli (numpy, absl, protobuf, ...) from
#    the hermetic pip repos used during the build. No `cp -n`: the Tizen
#    buildroot coreutils does not support it, and silencing errors here once
#    hid exactly that failure. Loud, verified copies only.
for dep in $BAZEL_ROOT/*/external/pypi_*/site-packages; do
    [ -d "$dep" ] || continue
    cp -a "$dep"/. $SITE_DIR/ || { echo "FATAL: failed to copy $dep"; exit 1; }
done
test -d $SITE_DIR/absl || { echo "FATAL: pypi deps missing after copy (absl)"; exit 1; }
test -d $SITE_DIR/numpy || { echo "FATAL: pypi deps missing after copy (numpy)"; exit 1; }

# 3b. Runtime-only deps missing from the bazel pip repos.
mkdir -p _extra_wheels
tar -xf %{SOURCE3003} -C _extra_wheels
for w in _extra_wheels/*.whl; do
    unzip -qo "$w" -d $SITE_DIR
done

# 4. tfcompile binary.
install -m 0755 bazel-bin/tensorflow/compiler/aot/tfcompile $AOT_ROOT/bin/tfcompile

# 5. saved_model_cli launcher at <prefix>/bin, where llvm's
#    TENSORFLOW_AOT_PATH/../../../../bin lookup expects it.
cat > $AOT_ROOT/bin/saved_model_cli << 'EOF'
#!/bin/sh
SELF="$(cd "$(dirname "$0")/.." && pwd)"
export PYTHONPATH="$SELF/lib/python%{py_ver}/site-packages${PYTHONPATH:+:$PYTHONPATH}"
exec "$SELF/python/bin/python%{py_ver}" -c "from tensorflow.python.tools.saved_model_cli import main; main()" "$@"
EOF
sed -i "s/%%{py_ver}/%{py_ver}/g" $AOT_ROOT/bin/saved_model_cli 2>/dev/null || true
chmod 0755 $AOT_ROOT/bin/saved_model_cli

# Smoke test: load TF and the AOT entry point through the bundled
# interpreter, exactly as the wrapper does. Run from a neutral directory —
# importing tensorflow with the source tree as cwd fails by design. Do NOT
# use `saved_model_cli --help`: absl prints usage and exits 1, so it can
# never be a pass/fail signal.
test -x %{buildroot}%{aot_prefix}/bin/saved_model_cli || \
    { echo "FATAL: saved_model_cli wrapper missing or not executable"; exit 1; }
( cd / && PYTHONPATH=%{buildroot}%{aot_prefix}/lib/python%{py_ver}/site-packages \
    %{buildroot}%{aot_prefix}/python/bin/python%{py_ver} -c \
    "import tensorflow as tf; from tensorflow.python.tools import saved_model_cli; print('smoke: TF', tf.__version__)" ) || \
    { echo "FATAL: saved_model_cli smoke test failed"; exit 1; }
