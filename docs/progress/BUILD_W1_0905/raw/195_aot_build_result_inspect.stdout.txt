# rules call it via `bash -c "realpath ..."`. Provide a shim.

# ... and no timeout(1) either; minimal shell replacement (rc!=0 on expiry).
cat > $PWD/shims/timeout << 'TMOEOF'
#!/bin/sh
secs=$1; shift
"$@" & pid=$!
( t=0; while [ $t -lt $secs ]; do sleep 10; t=$((t+10)); kill -0 $pid 2>/dev/null || exit 0; done; kill -9 $pid 2>/dev/null ) & wd=$!
wait $pid; rc=$?
kill $wd 2>/dev/null; wait $wd 2>/dev/null
exit $rc
TMOEOF
chmod +x $PWD/shims/realpath $PWD/shims/timeout
