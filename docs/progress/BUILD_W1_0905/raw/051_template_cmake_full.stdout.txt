set(PREBUILT_TF_XLA_RUNTIME
    "${CMAKE_CURRENT_SOURCE_DIR}/libtf_xla_runtime_prebuilt.a")
set(TF_XLA_RUNTIME_DUMMY_C
    "${CMAKE_CURRENT_BINARY_DIR}/tf_xla_runtime_dummy.c")

file(WRITE "${TF_XLA_RUNTIME_DUMMY_C}"
     "int __tf_xla_runtime_anchor(void){return 0;}\n")
add_library(tf_xla_runtime STATIC "${TF_XLA_RUNTIME_DUMMY_C}")
set_target_properties(tf_xla_runtime PROPERTIES LINKER_LANGUAGE C)
set_source_files_properties("${TF_XLA_RUNTIME_DUMMY_C}" PROPERTIES
  OBJECT_DEPENDS "${PREBUILT_TF_XLA_RUNTIME}")
add_custom_command(TARGET tf_xla_runtime POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${PREBUILT_TF_XLA_RUNTIME}"
          "$<TARGET_FILE:tf_xla_runtime>"
  COMMAND ${CMAKE_COMMAND} -E compare_files
          "${PREBUILT_TF_XLA_RUNTIME}"
          "$<TARGET_FILE:tf_xla_runtime>"
  COMMENT "Overwriting tf_xla_runtime with prebuilt PIC archive"
  VERBATIM)
