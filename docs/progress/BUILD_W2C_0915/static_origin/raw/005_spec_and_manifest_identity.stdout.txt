   300	prepare_package_list() {
   301	  local bins=""
   302	  local name=$1
   303	  shift
   304	  local executables="$@"
   305	  echo "Setting up $name for $executables"
   306	  ## Prepare the clang binaries separately
   307	  for executable in $executables; do
   308	    bins="$bins $executable `ldd $executable 2>/dev/null | sed -n 's,.*=> \(/[^ ]*\) .*,\1,p'`"
   309	  done
   310	
   311	  for b in $(echo "$bins" | tr ' ' '\n' | sort -u); do
   312	    local orig_file=$(readlink -f $b)
   313	    bins="$bins $orig_file"
   314	    if ! echo "$b" | grep -q '/\.\./'; then
   315	        # Ignore relative paths
   316	        echo $b >> $name.list
   317	    fi
   318	    echo $orig_file >> $name.list
   319	  done
   320	
   321	  ## Store the resulting list for packaging, setting up the right prefix, removing duplication of %emul_path if one exists
   322	  cat $name.list | \
   323	      sed -e 's|^|%{emul_path}|;s| |\n%{emul_path}|g' | \
   324	      grep -v "%{emul_path}%{emul_path}" | \
   325	      sort -u > %{_sourcedir}/$name-accel.list
   326	  binaries=$(echo "$binaries $bins" | tr ' ' '\n' | sort -u)
   327	}

   361	patch_binary() {
   362	  local binary=$1
   363	  local rtld=$2
   364	  # We don't need to accelerate shell scripts
   365	  local filetype=`file --brief $binary`
   366	  if echo "$filetype" | grep -q '^POSIX shell script'; then
   367	    return
   368	  fi
   369	
   370	  local outfile=${buildroot_emul_path}$binary
   371	  if ! test -e "$binary"; then
   372	    echo "WARNING: File not found, ignoring"
   373	    return
   374	  fi
   375	  if test -f "$outfile"; then
   376	    echo "WARNING: File exists, ignoring"
   377	    return
   378	  fi
   379	  mkdir -p ${outfile%/*}
   380	  cp -aL $binary $outfile
   381	  [ "$binary" = "$rtld" ] && return
   382	  [ "$binary" = "%{emul_path}$rtld" ] && return
   383	
   384	  # not all binaries have an .interp section
   385	  if patchelf --print-interpreter $outfile 1>/dev/null 2>/dev/null; then
   386	    patchelf --set-interpreter %{emul_path}$rtld $outfile
   387	  fi
   388	
   389	  # Only patch RUNPATH if one exists and is absolute
   390	  local rpath=$(readelf -d "$outfile" | sed -e '/RUNPATH/{s~.*\[\(.*\)\]~\1~;n};d')
   391	  if [ -n "$rpath" ]; then
   392	    first_char=$(echo "$rpath" | cut -c1)
   393	    if [ "$first_char" = "/" ]; then
   394	      patchelf --set-rpath "%{emul_path}%{_libdir}" $outfile
   395	    fi
   396	  fi
   397	}
   398	
   399	# Process all binaries directly (parallel jobs with simple file-based coordination)
   400	for binary in $binaries; do
   401	    patch_binary "$binary" "$LD" &
   402	done
   403	wait
