     1	#!/bin/bash
     2	#
     3	# This script creates builds the official release artifacts.
     4	#
     5	# Usage examples:
     6	#         ./scripts/create-assets.sh
     7	#         OUT=../out-dir ./scripts/create-assets.sh
     8	#
     9	
    10	ZSTDFLAGS="-19"
    11	
    12	function err() {
    13	  echo >&2 "$(tput setaf 1)ERROR: $*$(tput sgr0)"
    14	  exit 1
    15	}
    16	function info() {
    17	  echo "$(tput bold)$*$(tput sgr0)"
    18	}
    19	
    20	[[ -d tools ]] || err "'tools' directory not found, run script from bpftrace root dir"
    21	[[ -d man ]] || err "'man' directory not found, run script from bpftrace root dir"
    22	command -v zstd >/dev/null 2>&1 || err "zstd command not found, required for release"
    23	command -v asciidoctor >/dev/null 2>&1 || err "asciidoctor not found, required for manpage"
    24	
    25	# Take value from environment if set, otherwise use a tempdir
    26	: ${OUT:=$(mktemp -d)} || err "Failed to create temp dir"
    27	TMP="${OUT}/tmp"
    28	
    29	echo "Using '$OUT' as assert dir"
    30	
    31	set -e
    32	
    33	info "Creating tools archive"
    34	
    35	# bit of copying to avoid confusing tar flags not great but works
    36	mkdir -p "$TMP/bin"
    37	
    38	cp tools/*.bt "$TMP/bin"
    39	chmod +x "$TMP/bin/"*.bt
    40	tar --xz -cf "$OUT/tools.tar.xz" -C "$TMP/bin" "."
    41	
    42	info "Creating man archive"
    43	mkdir -p "$TMP/share/man/man8"
    44	cp man/man8/*.8 "$TMP/share//man/man8/"
    45	gzip -n "$TMP/share/man/man8/"*
    46	asciidoctor man/adoc/bpftrace.adoc  -b manpage -o - | gzip -n - > "$TMP/share/man/man8/bpftrace.8.gz"
    47	tar --xz -cf "$OUT/man.tar.xz" -C "$TMP/share" man
    48	
    49	info "Building bpftrace appimage"
    50	nix build .#appimage
    51	
    52	info "Creating bundle"
    53	cp ./result "$OUT/bpftrace"
    54	cp ./result "$TMP/bin/bpftrace"
    55	tar -cf "$OUT/binary_tools_man-bundle.tar" -C "$TMP" bin share
    56	zstd $ZSTDFLAGS -q -k "$OUT/binary_tools_man-bundle.tar"
    57	xz "$OUT/binary_tools_man-bundle.tar"
    58	
    59	echo "All assets created in $OUT"
    60	[[ -d "$TMP" ]] && rm -rf "$TMP"
