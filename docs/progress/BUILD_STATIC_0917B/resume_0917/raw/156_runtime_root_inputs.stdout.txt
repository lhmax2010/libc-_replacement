#!/bin/sh
# Global options must precede -t, whose following arguments belong to the tool.
expect_jobs=0
for arg do
  if [ "$expect_jobs" = 1 ]; then
    [ "$arg" = 1 ] || exit 64
    expect_jobs=0
    continue
  fi
  case "$arg" in
    -j|--jobs) expect_jobs=1 ;;
    -j1|--jobs=1) ;;
    -j*|--jobs=*) exit 64 ;;
  esac
done
[ "$expect_jobs" = 0 ] || exit 64
exec /tmp/ninja.real -j1 "$@"
