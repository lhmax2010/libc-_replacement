set -u -o pipefail
workspace=/home/toolchain/development/libc++_replacement
bin="$workspace/tmp/R67/probes/x86_64"
libs="$workspace/tmp/R67/build-x86_64/lib"
baseline="$workspace/tmp/GBS-ROOT/LIBCXX-2218-x86_64-20260806-c2/local/BUILD-ROOTS/scratch.x86_64.0/home/abuild/rpmbuild/BUILD/llvm-22.1.8/build/lib"
raw="$workspace/progress/R67/raw/x86_64/directed"
mkdir -p "$raw"
run_case() {
  local name=$1 binary=$2 mode=${3:-} library_path=${4:-} rc=0
  local log="$raw/$name.log"
  if [[ -n $library_path ]]; then
    printf 'COMMAND=env LD_LIBRARY_PATH=%q timeout 15s %q %q\n' "$library_path" "$binary" "$mode" > "$log"
    env LD_LIBRARY_PATH="$library_path" timeout 15s "$binary" ${mode:+"$mode"} >> "$log" 2>&1 || rc=$?
  else
    printf 'COMMAND=timeout 15s %q %q\n' "$binary" "$mode" > "$log"
    timeout 15s "$binary" ${mode:+"$mode"} >> "$log" 2>&1 || rc=$?
  fi
  printf 'EXIT_CODE=%d\n' "$rc" >> "$log"
  printf '%s\t%d\n' "$name" "$rc"
}
printf 'case\texit_code\n'
for mode in f1 f2 m1; do
  run_case "fixed_$mode" "$bin/r61_fixed" "$mode" "$libs:$baseline"
  run_case "reference_$mode" "$bin/r61_reference" "$mode"
done
run_case fixed_m2_lock "$bin/m2_lock_fixed" '' "$libs:$baseline"
run_case reference_m2_lock "$bin/m2_lock_reference"
run_case fixed_async_reference_return "$bin/async_reference_return_fixed" '' "$libs:$baseline"
run_case reference_async_reference_return "$bin/async_reference_return_reference"
