   218	inline TfLiteStatus AbsInt16EvalImpl(TfLiteContext* context, TfLiteNode* node,
   219	                                     TfLiteType expected_type) {
   220	  const TfLiteTensor* input;
   221	  TF_LITE_ENSURE_OK(context, GetInputSafe(context, node, 0, &input));
   222	  TfLiteTensor* output;
   223	  TF_LITE_ENSURE_OK(context, GetOutputSafe(context, node, 0, &output));
   224	  TF_LITE_ENSURE_TYPES_EQ(context, input->type, expected_type);
   225	  const int64_t num_elements = NumElements(input);
   226	  const int16_t* in_data = GetTensorData<int16_t>(input);
   227	  int16_t* out_data = GetTensorData<int16_t>(output);
   228	  for (int64_t i = 0; i < num_elements; ++i) {
   229	    out_data[i] = static_cast<int16_t>(
   230	        std::abs<int32_t>(static_cast<int32_t>(in_data[i])));
   231	  }
   232	  return kTfLiteOk;

   262	template <typename T>
   263	TfLiteStatus AbsEvalQuantized(TfLiteContext* context, TfLiteNode* node,
   264	                              TfLiteType type) {
   265	  const auto* op_data = static_cast<const OpData*>(node->user_data);
   266	  const int kMin = std::numeric_limits<T>::min();
   267	  const int kMax = std::numeric_limits<T>::max();
   268	
   269	  std::function<T(T)> func = [&](T i) {
   270	    const int32_t value = std::abs(i - op_data->input_offset);
   271	    if (!op_data->needs_rescale) {
   272	      return static_cast<T>(
   273	          std::min(std::max(value + op_data->output_offset, kMin), kMax));
   274	    }
   275	    const int32_t output = MultiplyByQuantizedMultiplier(
   276	                               value, op_data->multiplier, op_data->shift) +
   277	                           op_data->output_offset;
   278	    return static_cast<T>(std::min(std::max(output, kMin), kMax));
   279	  };
   280	
   281	  return EvalImpl<T>(context, node, func, type);
   282	}

   284	TfLiteStatus AbsEval(TfLiteContext* context, TfLiteNode* node) {
   285	  const TfLiteTensor* input = GetInput(context, node, 0);
   286	  const TfLiteType type = input->type;
