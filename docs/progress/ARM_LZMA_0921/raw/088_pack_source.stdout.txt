    26	static int rpmPackageFilesArchive(rpmfiles fi, int isSrc,
    27					  FD_t cfd, ARGV_t dpaths,
    28					  rpm_loff_t * archiveSize, char ** failedFile)
    29	{
    30	    int rc = 0;
    31	    rpmfi archive = rpmfiNewArchiveWriter(cfd, fi);
    32	
    33	    while (!rc && (rc = rpmfiNext(archive)) >= 0) {
    34	        /* Copy file into archive. */
    35		FD_t rfd = NULL;
    36		const char *path = dpaths[rpmfiFX(archive)];
    37	
    38		rfd = Fopen(path, "r.ufdio");
    39		if (Ferror(rfd)) {
    40		    rc = RPMERR_OPEN_FAILED;
    41		} else {
    42		    rc = rpmfiArchiveWriteFile(archive, rfd);
    43		}
    44	
    45		if (rc && failedFile)
    46		    *failedFile = xstrdup(path);
    47		if (rfd) {
    48		    /* preserve any prior errno across close */
    49		    int myerrno = errno;
    50		    Fclose(rfd);
    51		    errno = myerrno;
    52		}
    53	    }
    54	
    55	    if (rc == RPMERR_ITER_END)
    56		rc = 0;
    57	
    58	    /* Finish the payload stream */
    59	    if (!rc)
    60		rc = rpmfiArchiveClose(archive);
    61	
    62	    if (archiveSize)
    63		*archiveSize = (rc == 0) ? rpmfiArchiveTell(archive) : 0;
    64	
    65	    rpmfiFree(archive);
    66	
    67	    return rc;
    68	}
