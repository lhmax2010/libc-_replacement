import csv
import hashlib
from pathlib import Path

root = Path('docs/progress/W3')
rows = []
for path in sorted(p for p in root.rglob('*') if p.is_file() and p.name != 'INDEX.tsv'):
    data = path.read_bytes()
    rows.append((str(path), hashlib.sha256(data).hexdigest(), len(data)))
with (root / 'INDEX.tsv').open('w', newline='') as handle:
    writer = csv.writer(handle, delimiter='\t', lineterminator='\n')
    writer.writerow(('path', 'sha256', 'bytes'))
    writer.writerows(rows)
print(f'INDEX_ROWS={len(rows)}')
PY
