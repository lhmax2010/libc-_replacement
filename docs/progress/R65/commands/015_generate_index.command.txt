set -euo pipefail
index_tmp=docs/progress/R65/INDEX.tsv.tmp
printf 'path\tsha256\n' > "$index_tmp"
find docs/progress/R65 -type f ! -name INDEX.tsv ! -name INDEX.tsv.tmp -print0 | sort -z | while IFS= read -r -d '' path; do
  hash=$(sha256sum "$path" | awk '{print $1}')
  printf '%s\t%s\n' "$path" "$hash"
done >> "$index_tmp"
mv "$index_tmp" docs/progress/R65/INDEX.tsv
