def snapshot():
    records = []
    if not cg.exists():
        return

        for value in f.read_text().split():
            try:
                pid = int(value)

                records.append(dict(pid=pid, name=(proc/'comm').read_text().strip(),
                                    nice=os.getpriority(os.PRIO_PROCESS, pid),

def stop(reason):

    if cg.exists():
        # Queue termination before thaw so paused tasks cannot resume normal work.

                    pass
